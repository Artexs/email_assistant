import { Hono } from "hono";

import { auth, PORT } from "./auth";

const app = new Hono();

app.on(["POST", "GET"], "/api/auth/*", (c) => auth.handler(c.req.raw));

app.get("/api/me", async (c) => {
  const session = await auth.api.getSession({ headers: c.req.raw.headers });
  if (!session) return c.json({ error: "unauthorized" }, 401);
  return c.json({ userId: session.user.id, email: session.user.email, sessionId: session.session.id, expiresAt: session.session.expiresAt });
});

const server = Bun.serve({ port: PORT, hostname: "127.0.0.1", fetch: app.fetch });
console.log(`probe server on ${server.port}`);
