// Server-side auth.api calls (no HTTP): how does a CLI-created user get a first password?
// Run: cd /tmp/ea-s5-probe/ba && DISABLE_SIGNUP=1 bun probe2.ts ; DISABLE_SIGNUP=0 bun probe2.ts
import { readFileSync } from "node:fs";

import { betterAuth } from "better-auth";
import { Pool } from "pg";

const url = readFileSync("/tmp/ea-s5-probe/ba/.probe-url", "utf8").trim();
const pool = new Pool({ connectionString: url, max: 3 });
const disableSignUp = process.env.DISABLE_SIGNUP === "1";
let capturedUrl: string | null = null;
let capturedToken: string | null = null;

const auth = betterAuth({
  appName: "ea-probe",
  baseURL: "http://127.0.0.1:4123",
  secret: process.env.PROBE_SECRET ?? "set PROBE_SECRET (32+ random characters)",
  database: pool,
  trustedOrigins: ["http://127.0.0.1:4123"],
  emailAndPassword: {
    enabled: true,
    minPasswordLength: 12,
    disableSignUp,
    sendResetPassword: async ({ user, url, token }) => {
      capturedUrl = url;
      capturedToken = token;
      console.log(`  sendResetPassword called for user ${user.id} (url length ${url.length})`);
    },
  },
  user: { modelName: "auth_users", fields: { emailVerified: "email_verified", createdAt: "created_at", updatedAt: "updated_at" } },
  session: {
    modelName: "auth_sessions",
    fields: { userId: "user_id", expiresAt: "expires_at", ipAddress: "ip_address", userAgent: "user_agent", createdAt: "created_at", updatedAt: "updated_at" },
  },
  account: {
    modelName: "auth_accounts",
    fields: {
      userId: "user_id",
      accountId: "account_id",
      providerId: "provider_id",
      accessToken: "access_token",
      refreshToken: "refresh_token",
      idToken: "id_token",
      accessTokenExpiresAt: "access_token_expires_at",
      refreshTokenExpiresAt: "refresh_token_expires_at",
      createdAt: "created_at",
      updatedAt: "updated_at",
    },
  },
  verification: { modelName: "auth_verifications", fields: { expiresAt: "expires_at", createdAt: "created_at", updatedAt: "updated_at" } },
  advanced: { database: { generateId: "uuid" } },
});

const stamp = Date.now();
console.log(`== disableSignUp=${disableSignUp}`);

// A. server-side signUpEmail with a random password
try {
  const res = await auth.api.signUpEmail({ body: { email: `a-${stamp}@x.test`, password: crypto.randomUUID(), name: "A" } });
  console.log(`A signUpEmail server-side: ok, user ${res.user.id}`);
} catch (error) {
  console.log(`A signUpEmail server-side: refused: ${(error as { status?: string; body?: { code?: string } }).status ?? ""} ${(error as { body?: { code?: string } }).body?.code ?? (error as Error).message}`);
}

// B. a user row inserted by SQL (no credential account), then the reset-password flow
const email = `b-${stamp}@x.test`;
const inserted = await pool.query(
  `insert into auth_users (name, email, email_verified) values ($1, $2, true) returning id`,
  ["B Owner", email]
);
const userId = inserted.rows[0].id as string;
try {
  await auth.api.requestPasswordReset({ body: { email, redirectTo: "http://127.0.0.1:4123/setup" } });
  console.log(`B requestPasswordReset: ok; token captured: ${capturedToken !== null}; url starts with ${capturedUrl?.slice(0, 40)}`);
} catch (error) {
  console.log(`B requestPasswordReset: error ${(error as Error).message}`);
}
if (capturedToken) {
  try {
    await auth.api.resetPassword({ body: { newPassword: "correct-horse-battery-staple", token: capturedToken } });
    const accounts = await pool.query(`select provider_id, password is not null as has_hash from auth_accounts where user_id = $1`, [userId]);
    console.log(`B resetPassword: ok; accounts: ${JSON.stringify(accounts.rows)}`);
  } catch (error) {
    console.log(`B resetPassword: error ${(error as { body?: { code?: string } }).body?.code ?? (error as Error).message}`);
  }
  try {
    const signIn = await auth.api.signInEmail({ body: { email, password: "correct-horse-battery-staple" } });
    console.log(`B signInEmail after reset: ok, token length ${signIn.token?.length ?? 0}`);
  } catch (error) {
    console.log(`B signInEmail after reset: error ${(error as { body?: { code?: string } }).body?.code ?? (error as Error).message}`);
  }
  // C. the same token a second time must fail
  try {
    await auth.api.resetPassword({ body: { newPassword: "another-long-password-123", token: capturedToken } });
    console.log("C token reuse: ACCEPTED (bad)");
  } catch (error) {
    console.log(`C token reuse: refused (${(error as { body?: { code?: string } }).body?.code ?? (error as Error).message})`);
  }
}
await pool.end();
