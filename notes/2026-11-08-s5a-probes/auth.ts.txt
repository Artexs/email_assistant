import { betterAuth } from "better-auth";
import { Pool } from "pg";

import { urls } from "./db-urls";

// The D1 shape of the master plan, mapped onto our `users` table and the three `auth_*` tables.
export const BASE_URL = process.env.PROBE_BASE_URL ?? "http://127.0.0.1:4181";
export const DEV_ORIGIN = "http://localhost:4321";
export const PASSWORD = "correct-horse-battery-staple";
export const PASSWORD_2 = "another-long-passphrase-42";

export const events: string[] = [];
export const resetUrls: string[] = [];
export const logLines: string[] = [];

export function createAuth(opts: { baseURL?: string } = {}) {
  // Role `app`: proves that the app role reads and writes tables that `migrator` created.
  const pool = new Pool({ connectionString: urls.DATABASE_URL, max: 3 });
  const auth = betterAuth({
    appName: "ea-probe",
    baseURL: opts.baseURL ?? BASE_URL,
    secret: "probe-secret-not-a-real-secret-0123456789abcdef",
    database: pool,
    trustedOrigins: [BASE_URL, DEV_ORIGIN],
    emailAndPassword: {
      enabled: true,
      disableSignUp: true,
      minPasswordLength: 12,
      maxPasswordLength: 128,
      resetPasswordTokenExpiresIn: 60 * 60 * 24,
      revokeSessionsOnPasswordReset: true,
      sendResetPassword: async ({ url }) => {
        resetUrls.push(url);
      },
      onPasswordReset: async ({ user }) => {
        events.push(`onPasswordReset user=${user.id}`);
      },
    },
    user: {
      modelName: "users",
      fields: { name: "display_name", emailVerified: "email_verified", createdAt: "created_at", updatedAt: "updated_at" },
    },
    session: {
      modelName: "auth_sessions",
      fields: {
        userId: "user_id",
        expiresAt: "expires_at",
        ipAddress: "ip_address",
        userAgent: "user_agent",
        createdAt: "created_at",
        updatedAt: "updated_at",
      },
      expiresIn: 60 * 60 * 24 * 30,
      updateAge: 60 * 60 * 24,
    },
    account: {
      modelName: "auth_accounts",
      fields: {
        userId: "user_id",
        accountId: "account_id",
        providerId: "provider_id",
        accessToken: "access_token",
        refreshToken: "refresh_token",
        idToken: "id_token",
        accessTokenExpiresAt: "access_token_expires_at",
        refreshTokenExpiresAt: "refresh_token_expires_at",
        createdAt: "created_at",
        updatedAt: "updated_at",
      },
    },
    verification: {
      modelName: "auth_verifications",
      fields: { expiresAt: "expires_at", createdAt: "created_at", updatedAt: "updated_at" },
    },
    advanced: {
      cookiePrefix: "ea",
      database: { generateId: "uuid" },
      ipAddress: { ipAddressHeaders: ["x-client-ip"] },
    },
    rateLimit: { enabled: true, customRules: { "/sign-in/email": { window: 10, max: 3 } } },
    databaseHooks: {
      session: {
        create: {
          after: async (session, ctx) => {
            events.push(`session.create.after user=${session.userId} path=${ctx?.path ?? "none"}`);
          },
        },
      },
      account: {
        create: {
          after: async (account, ctx) => {
            events.push(`account.create.after provider=${account.providerId} path=${ctx?.path ?? "none"}`);
          },
        },
        update: {
          after: async (account, ctx) => {
            const sessionUser = (ctx as { context?: { session?: { user?: { id?: string } } } } | undefined)?.context?.session?.user?.id;
            events.push(
              `account.update.after keys=${Object.keys(account ?? {}).join("+")} typeof=${typeof account} path=${ctx?.path ?? "none"} sessionUser=${sessionUser ?? "none"}`
            );
          },
        },
      },
    },
    logger: {
      level: "warn",
      log: (level, message) => {
        logLines.push(`${level}: ${message}`);
        console.log(`[ba ${level}] ${message}`);
      },
    },
  });
  return { auth, pool };
}

export function tokenOf(url: string): string {
  const m = /\/reset-password\/([^?]+)\?/.exec(url);
  if (!m?.[1]) throw new Error(`no token in url (length ${url.length})`);
  return m[1];
}
