import { readFileSync } from "node:fs";

// The two URLs that `bun run db:worktree s5aprobe` printed (file mode 600; never print its content).
const lines = readFileSync(new URL("./.db-urls", import.meta.url), "utf8").split("\n");
const entries = lines
  .filter((line) => /^(DATABASE_URL|MIGRATOR_DATABASE_URL)=/.test(line))
  .map((line) => {
    const i = line.indexOf("=");
    return [line.slice(0, i), line.slice(i + 1).trim()] as const;
  });
export const urls = Object.fromEntries(entries) as { DATABASE_URL: string; MIGRATOR_DATABASE_URL: string };
if (!urls.DATABASE_URL || !urls.MIGRATOR_DATABASE_URL) throw new Error(".db-urls is incomplete");
