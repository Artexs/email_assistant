import { chromium } from "playwright-core";
const b = await chromium.launch({ executablePath: "/usr/bin/chromium" });
const p = await b.newPage();
const msgs = [];
p.on("console", (m) => msgs.push(`${m.type()}: ${m.text()}`));
p.on("pageerror", (e) => msgs.push(`pageerror: ${e.message}`));
await p.addInitScript(() => document.addEventListener("securitypolicyviolation", (e) => console.log(`CSPV ${e.violatedDirective} ${e.blockedURI} ${e.sample}`)));
await p.goto("http://localhost:4799/email_assistant/");
await p.waitForTimeout(800);
await p.click("button[data-open-modal]").catch((e) => msgs.push("noclick " + e.message));
await p.keyboard.type("docs");
await p.waitForTimeout(2500);
const res = await p.locator("dialog").innerText().catch(() => "nodialog");
console.log("RESULTS:", res.slice(0, 300).replace(/\n/g, " | "));
await p.goto("http://localhost:4799/email_assistant/pl/start/");
await p.waitForTimeout(500);
console.log(msgs.join("\n"));
await b.close();
