// Prints every CSS custom property definition (name: value) of a CSS file, one per line, sorted and unique.
const css = await Bun.file(process.argv[2]!).text();
const out = new Set<string>();
for (const m of css.matchAll(/(--[A-Za-z0-9_-]+)\s*:\s*([^;}]*)[;}]/g)) out.add(`${m[1]}: ${m[2]!.replace(/\s+/g, "")}`);
console.log([...out].sort().join("\n"));
